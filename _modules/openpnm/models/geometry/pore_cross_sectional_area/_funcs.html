@_geodocs
def sphere(
    network,
    pore_diameter='pore.diameter'
):
    r"""
    Calculate cross-sectional area assuming the pore body is a sphere

    Parameters
    ----------
    %(network)s
    %(Dp)s

    Returns
    -------
    areas : ndarray
        A numpy ndarry containing pore cross-sectional area values

    """
    D = network[pore_diameter]
    return _pi/4 * D**2
